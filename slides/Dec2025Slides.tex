\section{Model}
\begin{frame}{An incomplete markets economy}
\begin{itemize}
    \item Time is discrete and infinite.
    \item A continuum of households, each with 
    \item[-] $\overline{n}$ unit of indivisible labor
    \item[-] idiosyncratic productivity shock $z$:
    \begin{equation}
    \label{eq:z}
    \ln z'=\rho_z\ln z + \varepsilon_z, \varepsilon_z \overset{\mathrm{iid}}{\sim} N(0,\sigma_z^2)
    \end{equation}
    \item Households cannot issue any asset contingent on future idiosyncratic shock.
    \item Households can invest in physical capital $a$ and human capital $h$
\end{itemize}
    
\end{frame}

\begin{frame}{Production technology}
\begin{itemize}
\item[]
\begin{equation}
    F(K,L)=K^{1-\alpha} L^{\alpha},
\end{equation}
\begin{equation}
    K=\textstyle\int a_idi, \quad L=\textstyle\int n_iz_ix(h_i)di
\end{equation}
    \item $n_i \in \{0,\overline{n}\}$: labor supply decision
    \item Sectoral productivity for effective labor units
    \begin{equation}
    \label{eq:x}
x(h)=\left \{ 
\begin{array}{cl}
1-\lambda  & \text{low sector if }h<h_{M} \\ 
1 & \text{middle sector if }h_{M}<h<h_{H} \\ 
1+\lambda  & \text{high sector if }h>h_{H}%
\end{array}%
\right. 
\end{equation}
\item Anticipated AI introduction: $\gamma \in (0,1)$
\begin{equation}
    \label{eq:xAI}
x(h')=\left \{ 
\begin{array}{cl}
1-\lambda + \gamma \lambda & \text{low sector if }h'<h_{M} \\ 
1 & \text{middle sector if }h_{M}<h'<h_{H} \\ 
1+\lambda + \gamma \lambda & \text{high sector if }h'>h_{H}%
\end{array}%
\right. 
\end{equation}
\end{itemize}

\end{frame}

\begin{frame}{Household's Problem}
\begin{itemize}
\item Based on $\color{blue}z_t\color{black}$
\begin{align}
\max_{\left \{ c_t,a_{t+1},n_t,e_t\right \}_{t=0}^{\infty }}&E_0\left[
\sum\limits_{t=0}^{\infty }\beta^t( \ln c_t-\chi_n n_t-\chi_e e_t ) \right] \text{ s.t.} \\
    c_t+a_{t+1}&=n_t( w_t\color{blue}z_t\color{black}x(h_t))+(1+r_t)a_t \\
    c_t& \geq 0 \text{ and } a_{t+1} \geq 0 \\
    e_t& \in \{0, e_L, \color{red}\mathbf{1}_{n_t=0}\color{black}e_H\} \\
    h_{t+1}&=\color{blue}z_t\color{black}e_t + (1-\delta)h_t
\end{align}    
\item $z_t$: productivity for both work and human capital accumulation
\item Take factor prices as given: $w_t = F_L(K,L)$ and $r_t=F_K(K,L)$.
\end{itemize}
\end{frame}

\section{Household Decisions in a Two-Period Model}
\begin{frame}{Two-period model household decisions}
\begin{itemize}
    \item Period 2: work $n=1$ iff $z>\overline{z}(h,a)$
    \medskip
    \item Period 1: states $\{z,h,a\}$ map to decisions $n\in\{0,1\}, e\in\{0,e_L,e_H\}, a'$
    \smallskip
    \item[] Additive separability of $\{n,e\}$ and $c$ in utility $\Rightarrow$ 
    \item[] Step 1: $\{n,e\}$ conditional on optimal $a'$ without uncertainty
    \item[] Step 2: $\{n,e,a'\}$ affected by uncertainty
    \item[] Step 3: $a'$ with uncertainty conditional on $\{n,e\}$ 
\end{itemize}

\end{frame}

\begin{frame}{Effect of idiosyncratic risk $z'$ on $\{n,e,a'\}$}
\begin{proposition}
    The uninsured idiosyncratic risk in $z'$ makes households in period 1 save more, work more and invest less in human capital.
\end{proposition}
\begin{itemize}
    \item Idiosyncratic risk raises saving and labor supply (Swanson 2012)
    \item Idiosyncratic risk lowers return to human capital
    \item[-] higher $x'$ $\rightarrow$ $ln(c')$ more concave in $z'$
    \item[-] higher $x'$ $\rightarrow$ more exposure to low $z'$ risk due to endogenous labor supply
    
\end{itemize}

\end{frame}

\begin{frame}{Effect of part-time learning on saving $a'$}
Normalize $z=1$ and $\ln z' \sim \mathcal{N}(0,\sigma_z^2)$
\begin{equation*}\label{eq:savingprob}
\max_{a'} : \ln(c) + \beta \mathbb{E}_{z'}[\ln(c')],
\end{equation*}
subject to:
\begin{equation}
c + a' = a + x , \quad c' = a' + txz'
\end{equation}
The second-order approximate solution to the FOC is:
\begin{equation}
a'^*(x,a;t) = \underbrace{\frac{\beta(a+x) - tx\mu}{1+\beta}}_{\text{CE}} + \underbrace{\frac{t^2x^2\Sigma}{\beta(a+x + tx\mu)}}_{\text{Precautionary}}
\end{equation}
where $\mu\equiv \mathbb{E}[z']$ and $\Sigma \equiv\operatorname{Var}(z')$


\end{frame}

\begin{frame}{Effect of part-time learning on saving $a'$}
Compare optimal $a'$ between $e=e_L$ $(t>1)$ and $e=0$ $(t=1)$
\begin{equation} \label{eq:dadt_integral}
    \Delta_{\text{part-time}}(x,a;t) = a'^*(x,a;t) - a'^*(x,a;1) = \int_1^t \frac{\partial a'^*}{\partial u}(x,a;u) du,
\end{equation}
Part-time learning \textcolor{red}{\textbf{crowds out}} saving for \textcolor{red}{\textbf{low-income}} households and \textcolor{blue}{\textbf{crowds in}} saving for \textcolor{blue}{\textbf{high-income}} households.
\begin{proposition}\label{prop:dadt}
    If $\frac{\Sigma}{\mu}>\sigma^*(t)$, $\Delta_{\text{part-time}}(x,a;t)<0$ for $x<x^\ast(a,t)$, and $\Delta_{\text{part-time}}(x,a;t)>0$ for $x>x^\ast(a,t)$.
\end{proposition}

\end{frame}

\begin{frame}{Effect of full-time training on saving $a'$}
\begin{itemize}
    \item Full-time training requires $n=0$.
    \item Household forgoes period-1 labor income.
\end{itemize}
\begin{align}
\max_{a'} &: \ln(c) + \beta \mathbb{E}_{z'}[\ln(c')],\notag\\
e=e_H&: \quad c + a' = a, \quad c' = a' + txz'\\
e=0&: \quad c + a' = a + x, \quad c' = a' + xz'
\end{align}
\end{frame}

\begin{frame}{Effect of full-time training on saving $a'$}
Compare optimal $a'$ between $e=e_H$ $(t>1,n=0)$ and $e=0$ $(t=1)$
\begin{align}
    \Delta_{\text{full-time}}(x,a;t) &=\Delta_{\text{part-time}}(x,a;t) + \Delta_H(x,a;t) \notag \\
    \text{where } \Delta_H(x,a;t) &\equiv x \Bigg[ -\frac{\beta}{1+\beta} + \frac{\Sigma}{\beta} \frac{t^2 x^2}{(a + x + t x \mu)(a + t x \mu)} \Bigg]
\end{align}
Full-time training \textcolor{red}{\textbf{crowds out}} saving for \textcolor{red}{\textbf{low-income}} households and \textcolor{blue}{\textbf{crowds in}} saving for \textcolor{blue}{\textbf{high-income}} households.
\begin{proposition}\label{prop:fulltime}
If $\frac{\Sigma}{\mu}>\max\{\sigma^*(t),\hat{\sigma}(t)\}$, $\Delta_{\text{full-time}}(x,a;t)<0$ for $x<\min\{x^*(a,t),\hat{x}(a,t)\}$, $\Delta_{\text{full-time}}(x,a;t)>0$ for $x>\max\{x^*(a,t),\hat{x}(a,t)\}$.
\end{proposition}

\end{frame}

\begin{frame}{Effects of an Anticipated Period-2 AI Shock}
\begin{itemize}
    \item on human capital investment:
    \item[-] $ e \downarrow$ for $h < h_M/(1-\delta)$, $ e \uparrow$ for $h > h_M/(1-\delta)$
    \medskip
    \item on labor supply:
    \item[-] via income effect: $ n \downarrow$ for low- and high- sectors 
    \item[-] via full-time training: $n \uparrow$ for $h < h_M/(1-\delta)$, $ n \downarrow$ for $h > h_M/(1-\delta)$
    \medskip
    \item on saving:
    \item[-] $ a' \downarrow$ for low sector, $ a' \uparrow$ for high sector
    \item[-] ambiguous for middle sector
    
\end{itemize}
\end{frame}

\begin{frame}{Limitations of the two-period model}

\begin{itemize}
    \item General Equilibrium Effects
    \item[-] wage and interest rate respond and shape household decisions
    \medskip
    \item Composition Effects
    \item[-] human capital adjustment $\Rightarrow$ shifting composition of labor across sectors
    \medskip
    \item Transition Dynamics
    \item[-] human capital takes time to build
    \item[-] effects of anticipating AI and responding to AI
\end{itemize}
\medskip
A quantitative model with infinite horizon, endogenous asset accumulation, and GE. 
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusions}

How AI-driven technological change affects human capital investment and, in turn, aggregate outcomes and economic inequality?
\smallskip
\begin{itemize}
\item A quantitative incomplete markets model with endogenous HC investment, sectoral structure, transition dynamics, and GE feedback
\medskip 
\item AI induces \textbf{job polarization} via human capital adjustment
\medskip
\item Human capital adjustment is crucial in determining AI's economic impacts 
\item[-] \textbf{reduces} labor supply
\item[-] \textbf{reverses} AI's positive effect on income and consumption inequalities
\item[-] \textbf{mitigates} AI's negative effect on wealth inequality
    
\end{itemize}
    
\end{frame}
